import pyspark.sql.functions as F
from pyspark.sql import SparkSession

# Build dynamic pipeline
spark = SparkSession.builder \
    .appName("BishnuPortfolio") \
    .getOrCreate()

data = spark.read.json("source_logs/*.json")
cleaned = data.filter(F.col("status") == "SUCCESS") \
              .groupBy("category") \
              .agg(F.count("*").alias("total"))

cleaned.write.mode("overwrite").saveAsTable("metrics")
print("🚀 Data pipeline execution complete!")
